\def\circuitscale{0.92}

\begin{document}
\begin{circuitikz}[emt diagram, scale=\circuitscale, transform shape]
\ctikzset{bipoles/length=1.05cm}

% Winding resistances, leakage inductance, and ideal off-nominal ratio.
\coordinate (e1) at (2.0,2.8);
\coordinate (e2) at (5.1,2.8);
\node[transformer core, anchor=A1] (tap) at (6.8,2.8) {};
\draw[semithick] (-0.2,2.8) to[R,l=$R_1$] (e1)
  to[L,l=$X$] (e2) -- (tap.A1);
\draw[semithick] (tap.B1) to[R,l=$R_2$] ++(2.0,0)
  coordinate (terminal2);
\node[sig, above] at (tap.north) {$\tau:1$};
\draw[semithick] (tap.A2) -- (tap.A2 |- 0,\returnrow);
\draw[semithick] (tap.B2) -- (tap.B2 |- 0,\returnrow)
  node[ground] {};

% Each core branch has a loss conductance and nonlinear inductance.
\foreach \x/\e in {2.0/1,5.1/2} {
  \draw[semithick] (\x,2.8) -- (\x,1.8);
  \draw[semithick] (\x,1.8) -- ++(-0.75,0)
    to[R] ++(0,-2.4)
    -- (\x,-0.6);
  \draw[semithick] (\x,1.8) -- ++(0.75,0)
    to[L] ++(0,-2.4)
    -- (\x,-0.6) -- (\x,\returnrow);
  \node[sig, anchor=north] at (\x-0.75,-0.68) {$\beta_\e G_\mathrm{c}$};
  \node[sig, anchor=north] at (\x+0.75,-0.68) {$\beta_\e g$};
  \node[signal, label={[sig]above:$e_\e$}] at (\x,2.8) {};
  \node[signal] at (\x,1.8) {};
  \node[signal] at (\x,-0.6) {};
  \node[signal] at (\x,\returnrow) {};
}
\draw[semithick] (2.0,\returnrow) -- (tap.A2 |- 0,\returnrow);
\node[ground] at (3.55,\returnrow) {};

% Winding terminals and outward current directions.
\node[signal, label={[sig]above:$v_{\mathrm{w},1}$}] (terminal1)
  at (-0.2,2.8) {};
\node[signal, label={[sig]above:$v_{\mathrm{w},2}$}] at (terminal2) {};
\draw[->, line] (terminal1) -- ++(-1.0,0)
  node[sig, below] {$-i_{\mathrm{w}1}$};
\draw[->, line] (terminal2) -- ++(1.0,0)
  node[sig, below] {$-i_{\mathrm{w}2}$};

  node[sig, midway, above] {$i_{12}$};
