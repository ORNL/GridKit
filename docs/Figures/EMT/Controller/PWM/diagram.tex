  at ($(origin)+(1.3,\striprow)$) {$v_{\mathrm{dc}}\mathcal{L}_u^{-1/2}$};
\node[signal] (w) at ($(origin)+(3.1,\striprow)$) {};
\node[block, minimum width=1.3cm] (gain)
  at ($(origin)+(4.1,\striprow)$) {$\dfrac{2}{v_{\mathrm{dc}}}$};
\node[block, minimum width=1.6cm] (park)
  at ($(origin)+(6.0,\striprow)$) {$\mathbf{T}^{\mathsf{T}}(\theta)$};
\node[block, minimum width=2.4cm] (duty)
  at ($(origin)+(8.6,\striprow)$) {$d_\ell=\dfrac{1+m_\ell}{2}$};
\node[block, minimum width=2.2cm] (switching)
  at ($(origin)+(11.5,\striprow)$) {$S_\mu(t,d_\ell)$};
\draw[->, line] ($(origin)+(-0.6,\striprow)$)
  -- node[sig, above, pos=0.3] {$\mathbf{u}$} (limit.west);
\draw[line] (limit.east) -- (w);
\draw[->, line] (w) -- (gain.west);
\draw[->, line] (gain.east) -- node[sig, above] {$\mathbf{m}$} (park.west);

\draw[->, line] (duty.east) -- node[sig, above] {$d_\ell$} (switching.west);
