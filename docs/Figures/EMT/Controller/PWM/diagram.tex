\def\modulationIndex{0.8}
\def\modulationFrequency{60}
\def\carrierFrequency{900}
\def\topScale{216000}
\def\bottomScale{900}

\tikzset{declare function={
  sigmaS(\x,\m) = ifthenelse(\x>=0,
    1/(1+exp(-\m*\x)), exp(\m*\x)/(1+exp(\m*\x)));
  st(\x) = \x-floor(\x);
  taum(\x,\q) = \x-(st(\carrierFrequency*\x)-\q)/\carrierFrequency;
  da(\x) = (1+
    \modulationIndex*sin(360*\modulationFrequency*\x))/(2*\carrierFrequency);
  db(\x) = (1+
    \modulationIndex*sin(360*\modulationFrequency*\x-120))/(2*\carrierFrequency);
  dc(\x) = (1+
    \modulationIndex*sin(360*\modulationFrequency*\x+120))/(2*\carrierFrequency);
  pa(\x,\q,\m) = sigmaS(\x-taum(\x,\q),\m)
    -sigmaS(\x-taum(\x,\q)-da(taum(\x,\q)),\m);
  pb(\x,\q,\m) = sigmaS(\x-taum(\x,\q),\m)
    -sigmaS(\x-taum(\x,\q)-db(taum(\x,\q)),\m);
  pc(\x,\q,\m) = sigmaS(\x-taum(\x,\q),\m)
    -sigmaS(\x-taum(\x,\q)-dc(taum(\x,\q)),\m);
  sa(\x,\m) = pa(\x,-1,\m)+pa(\x,0,\m)+pa(\x,1,\m);
  sb(\x,\m) = pb(\x,-1,\m)+pb(\x,0,\m)+pb(\x,1,\m);
  sc(\x,\m) = pc(\x,-1,\m)+pc(\x,0,\m)+pc(\x,1,\m);
}}
