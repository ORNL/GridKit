% TUNABLES: gain width and spacing of the two lower branches.
\def\gainwidth{1.35cm}
\def\integralrow{-2.2}
\def\trackingrow{-4.2}

\begin{document}
\begin{tikzpicture}[emt diagram]

% Voltage error and proportional path.
\node[sum] (error) at (3,0) {$\Sigma$};
\node[block, minimum width=\gainwidth] (Kp) at (5.1,0) {$K_P$};
\node[sum] (command) at (7.2,0) {$\Sigma$};
\draw[->, line] (error.east) -- (Kp.west);
\draw[->, line] (Kp.east) -- (command.west);
\node[pm, above left=0pt and 1pt of command.west] {$+$};

% Integral path, with anti-windup added before integration.
\node[block, minimum width=\gainwidth] (Ki)
    at (3,\integralrow) {$K_I$};
\node[sum] (rate) at (5.1,\integralrow) {$\Sigma$};
\node[block, minimum width=\gainwidth] (integrator)
    at (7.2,\integralrow) {$\dfrac{1}{s}$};
\draw[->, line] (error.south) -- (Ki.north);
\draw[->, line] (Ki.east) -- (rate.west);
\node[pm, above left=0pt and 1pt of rate.west] {$+$};
\draw[->, line] (rate.east) -- (integrator.west);
\draw[->, line] (integrator.north) -- (command.south);
\node[pm, below right=1pt and 0pt of command.south] {$+$};

% Grid-current feedforward and capacitor-current decoupling.
\node[block, minimum width=2.6cm]
    (feedforward) at (7.2,1.8)
    {$\mathbf{i}_g+\mathbf{J}\omega C\mathbf{v}$};
\draw[->, line] (feedforward.south) -- (command.north);
\node[pm, above right=1pt and 0pt of command.north] {$+$};

% Total current command supplied to the inner current controller.
\node[signal] (icmd) at (9.1,0) {};
\draw[line] (command.east) -- (icmd);
\draw[->, line] (icmd) -- node[sig, above, pos=0.75] {$\mathbf{i}^{\mathrm{cmd}}$} (11.8,0);

% Track the current limit owned by the inner current controller.
\node[sum] (tracking) at (9.1,\trackingrow) {$\Sigma$};
\node[block, minimum width=\gainwidth] (Kaw)
    at (5.1,\trackingrow) {$K_{\mathrm{aw}}$};
\draw[->, line] (icmd) -- (tracking.north);
\node[pm, above left=1pt and 0pt of tracking.north] {$-$};
\draw[->, line] (tracking.west) -- (Kaw.east);
\draw[->, line] (Kaw.north) -- (rate.south);
\node[pm, below left=1pt and 0pt of rate.south] {$+$};

% External references and measured voltage enter directly at the error sum.
\node[sig] (vref) at (0,0) {$\mathbf{v}^{\mathrm{ref}}$};
\node[sig] (v) at (3,4.4) {$\mathbf{v}$};
\node[sig] (omega) at (7.2,4.4) {$\omega$};
\node[sig] (ig) at (8,4.4) {$\mathbf{i}_g$};
\node[signal] (vsplit) at (3,3) {};
\draw[->, line] (vref.east) -- (error.west);
\node[pm, above left=0pt and 1pt of error.west] {$+$};
\draw[->, line] (v.south) -- (error.north);
\node[pm, above right=1pt and 0pt of error.north] {$-$};
\draw[->, line] (vsplit) -- (6.4,3) -- (feedforward.north -| 6.4,0);
\draw[->, line] (omega.south) -- (feedforward.north -| omega);
\draw[->, line] (ig.south) -- (feedforward.north -| ig);
\draw[->, line] (11.8,\trackingrow) -- node[sig, above, pos=0.25]
    {$\mathbf{i}^{\mathrm{lim}}$} (tracking.east);
\node[pm, above right=0pt and 1pt of tracking.east] {$+$};

% Enclose the model-owned states, variables, and feedback paths.
\begin{scope}[on background layer]
    \node[operator enclosure, densely dotted,
        fit=(error)(Kp)(Ki)(rate)(integrator)
            (feedforward)(command)(icmd)(tracking)(Kaw)(vsplit)] {};
\end{scope}

\end{tikzpicture}
\end{document}
