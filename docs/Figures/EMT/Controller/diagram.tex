% Shared body for the inverter wiring figures. The wrappers diagram_gfl.tex
% and diagram_gfm.tex set \gfmfalse or \gfmtrue before inputting this file;
% the switch selects the outer controller, its setpoint inputs, and the extra
% measurement wiring for the outer-control variants.

% TUNABLES: control-row height, frame column, and setpoint column.
\def\framecolumn{6.8}
\def\controlrow{-4.4}
\def\setpointcolumn{18.6}
\definecolor{measurementblue}{HTML}{0072B2}
\ifgfm
\usepackage{circuitikz}
\ctikzset{bipoles/thickness=1}
\input{control-symbols.tex}
\fi

\begin{document}
\begin{tikzpicture}[emt diagram,
    block/.append style={minimum height=1.25cm, align=center},
    measurement/.style={line, dashed, draw=measurementblue}]

% ================= DC supply and bridge =================
\node[minimum width=2.6cm, minimum height=1.9cm] (bridge) at (8.8, 0) {};
% DC/AC converter glyph with separate voltage-measurement and switching ports.
\node[circle, draw, semithick, minimum size=1.5cm, inner sep=0pt]
    (converter) at (bridge.center) {};
\begin{scope}[shift={(bridge.center)}]
    \draw[semithick] (converter.south west) -- (converter.north east);
    \draw[semithick] (-0.47,0.32) -- (-0.07,0.32);
    \draw[semithick, dash pattern=on 1.2pt off 0.8pt]
        (-0.47,0.22) -- (-0.07,0.22);
    \draw[semithick] (0.05,-0.27)
        sin (0.16,-0.16) cos (0.27,-0.27)
        sin (0.38,-0.38) cos (0.49,-0.27);
    \coordinate (vdc) at (-0.4,-0.63443);
    \coordinate (gates) at (0.4,-0.63443);
\end{scope}

% ================= LCL filter and terminal =================
% GFM: retain the capacitor, resistor centers, and current-measurement locations.
\node[signal] (voabc) at (\ifgfm 14.15\else 12.65\fi, 0) {};
\ifgfm
\node[minimum width=1.35cm, minimum height=0.65cm]
    (zs) at ($(voabc)+(-3,0)$) {};
\node[minimum width=1.35cm, minimum height=0.65cm]
    (zg) at ($(voabc)+(3,0)$) {};
\draw[semithick] (zs.west) to[american resistor] (zs.east);
\draw[semithick] (zg.west) to[american resistor] (zg.east);
\node[sig] at ($(zs.center)+(0,0.5)$) {$\mathbf R_{\mathrm{s}}$};
\node[sig] at ($(zg.center)+(0,0.5)$) {$\mathbf R_g$};
\else
\node[block, minimum width=1.35cm, minimum height=0.65cm]
    (zs) at ($(converter.east)!0.5!(voabc)$) {$\mathbf Z_{\mathrm{s}}$};
\node[block, minimum width=1.35cm, minimum height=0.65cm]
    (zg) at (14.15, 0) {$\mathbf Z_g$};
\fi
\node[signal] (tbus) at (\ifgfm 19.65\else 16.15\fi, 0) {};
\draw[line] (converter.east) -- (zs.west);
\ifgfm
\draw[semithick] (zs.east) -- (12.15,0)
    to[american inductor] (voabc)
    to[american inductor] (16.15,0) -- (zg.west);
\node[sig] at ($(voabc)+(-1,0.5)$) {$\mathbf L_{\mathrm{s}}$};
\node[sig] at ($(voabc)+(1,0.5)$) {$\mathbf L_g$};
\else
\draw[line] (zs.east) -- (voabc) -- (zg.west);
\fi
\draw[line] (zg.east) -- (tbus);
\ifgfm
% Ideal disconnect at the PCC, using the established Switch contact symbol.
\node[signal] (pccleft) at (20.75,0) {};
\node[signal] (pcc) at (21.55,0) {};
\draw[line] (tbus) -- (pccleft);
\draw[line] (pccleft) -- (21.4,0.4);
\node[sig, above right=2pt and 2pt of pcc] {PCC};
\draw[->, line] (pcc) -- (25.6,0) node[sig, right] {Bus};
\else
\draw[->, line] (tbus) -- (18.6,0) node[sig, right] {Bus};
\fi
% Compact single-line representation of the three-phase shunt capacitor.
\begin{scope}[shift={(voabc)}]
    \draw[line] (0,0) -- (0,0.65);
    \draw[semithick] (-0.27,0.65) -- (0.27,0.65);
    \draw[semithick] (-0.27,0.8) -- (0.27,0.8);
    \node[sig, anchor=west] at (0.37,0.725) {$\mathbf C$};
    \draw[line] (0,0.8) -- (0,1.15);
    \draw[semithick] (-0.27,1.15) -- (0.27,1.15);
    \draw[semithick] (-0.18,1.27) -- (0.18,1.27);
    \draw[semithick] (-0.09,1.39) -- (0.09,1.39);
    \coordinate (cground) at (0,1.39);
\end{scope}
% Current sensors encircle the conductors without electrical junction dots.
\draw[measurement] (12.15,0) ellipse[x radius=0.13, y radius=0.26];
\coordinate (iabc) at (12.15,-0.26);
\draw[measurement] (\ifgfm 16.15\else 15.65\fi,0)
    ellipse[x radius=0.13, y radius=0.26];
\coordinate (igabc) at (\ifgfm 16.15\else 15.65\fi,-0.26);

% ================= Reference frame and control =================
\node[block] (icc) at (12.4, \controlrow) {Current\\Control};
\node[block] (outer) at (15.9, \controlrow) {\ifgfm Voltage\else Power\fi\\Control};

% ================= Phase-domain measurements =================
% Reference-frame transformations are implicit at the controller boundary.
\draw[->, measurement] (iabc) -- ([xshift=-2.5mm]icc.north);
\node[sig, text=measurementblue, anchor=east] at (12.0, -1.2) {$\mathbf{i}$};
\ifgfm
\node[sig, text=measurementblue, anchor=west] at (14.3, -1.2) {$\mathbf v_{\mathrm{o}}$};
\else
\draw[->, measurement] (voabc) -- ([xshift=2.5mm]icc.north);
\node[sig, text=measurementblue, anchor=west] at (12.8, -1.2) {$\mathbf v_{\mathrm{o}}$};
\fi
\draw[->, measurement] (igabc) -- ([xshift=\ifgfm 2.5mm\else -2.5mm\fi]outer.north);
\ifgfm
\node[sig, text=measurementblue, anchor=west] at (16.3,-1.2) {$\mathbf i_g$};
\else
\node[sig, text=measurementblue, anchor=east] at (15.5,-1.2) {$\mathbf i_g$};
\fi

\ifgfm
% ================= Outer voltage-control wiring =================
% Branch one-quarter of the measurement-arrow height above current control.
\coordinate (voicc) at ([xshift=2.5mm]icc.north);
\coordinate (voovc) at ([xshift=-2.5mm]outer.north);
\coordinate (vbranchheight) at ($(voicc)!0.25!(voabc)$);
\node[signal, fill=measurementblue] (vouter) at (voabc |- vbranchheight) {};
\draw[measurement] (voabc) -- (vouter);
\draw[->, measurement] (vouter) -- (vouter -| voicc) -- (voicc);
\draw[->, measurement] (vouter) -- (vouter -| voovc) -- (voovc);
\fi

% ================= Control chain =================
\ifgfm
% Show the current-control input limiter with the shared saturation glyph.
\begin{scope}[shift={(14.15,\controlrow)}, xscale=0.55, yscale=-0.55, transform shape]
    \staticlimit{currentlimit}{(0,0)}{}{}
\end{scope}
\draw[line] (outer.west) -- (currentlimit.east);
\draw[->, line] (currentlimit.west) -- (icc.east);
\node[sig, below] at (14.15,\controlrow-0.35) {$\mathbf{i}^{\mathrm{cmd}}$};
\else
\draw[->, line] ([yshift=-2.5mm]outer.west)
    -- node[sig, below] {$\mathbf{i}^{\mathrm{cmd}}$} ([yshift=-2.5mm]icc.east);
\draw[->, line] ([yshift=2.5mm]icc.east)
    -- node[sig, above] {$\mathbf{i}^{\mathrm{lim}}$} ([yshift=2.5mm]outer.west);
\fi

% ================= PWM =================
% The modulator sits under the bridge inside the dq frame: it limits the
% voltage command, returns the limited command, and sends the switching
% functions straight up to the bridge.
\node[block] (pwm) at (8.8, \controlrow) {PWM};
\draw[->, measurement] ([xshift=4mm]pwm.north) -- (gates);
\node[sig, text=measurementblue, anchor=west] at (9.35,-1.2) {$\mathbf{s}$};
\draw[->, line] ([yshift=-2.5mm]icc.west)
    -- node[sig, below] {$\mathbf{u}^{\mathrm{cmd}}$} ([yshift=-2.5mm]pwm.east);
\draw[->, line] ([yshift=2.5mm]pwm.east)
    -- node[sig, above] {$\mathbf{u}^{\mathrm{lim}}$} ([yshift=2.5mm]icc.west);

% ================= Inverter-terminal voltage measurement =================
% GFL power control measures terminal voltage; GFM uses it in primary control.
\ifgfm\else
\draw[->, measurement] (tbus)
    -- ([xshift=2.5mm]outer.north);
\node[sig, text=measurementblue, anchor=west]
    at (16.3,-1.2) {$\mathbf v_{\mathrm{t}}$};
\fi

% The DC-terminal voltage is measured by the modulator.
\draw[->, measurement] (vdc) -- ([xshift=-4mm]pwm.north);
\node[sig, text=measurementblue, anchor=east] at (8.25,-1.2) {$v_{\mathrm{dc}}$};

% ================= Setpoints =================
% GFM primary and plant control are functional blocks in the general architecture.
\ifgfm
\node[block] (primary) at (19.4, \controlrow) {Primary\\Control};
\node[block] (plant) at (22.9, \controlrow) {Plant\\Control};
\coordinate (controlextent) at (plant.south east);
% Primary control measures inverter-terminal power before the PCC disconnect.
\draw[measurement] (19.15,0) ellipse[x radius=0.13, y radius=0.26];
\draw[->, measurement] (19.15,-0.26) -- ([xshift=-2.5mm]primary.north);
\draw[->, measurement] (tbus) -- ([xshift=2.5mm]primary.north);
\node[sig, text=measurementblue, anchor=east] at (19.0,-1.2) {$\mathbf i_g$};
\node[sig, text=measurementblue, anchor=west] at (19.8,-1.2) {$\mathbf v_{\mathrm{t}}$};
\draw[->, line] (primary.west)
    -- node[sig, above] {$\mathbf v^{\mathrm{ref}}$} (outer.east);
% Plant control regulates grid exchange at the PCC.
\draw[measurement] (22.65,0) ellipse[x radius=0.13, y radius=0.26];
\node[signal] (vplant) at (23.15,0) {};
\draw[->, measurement] (22.65,-0.26) -- ([xshift=-2.5mm]plant.north);
\draw[->, measurement] (vplant) -- ([xshift=2.5mm]plant.north);
\node[sig, text=measurementblue, anchor=east] at (22.5,-1.2) {$\mathbf i_{\mathrm{PCC}}$};
\node[sig, text=measurementblue, anchor=west] at (23.3,-1.2) {$\mathbf v_{\mathrm{PCC}}$};
\draw[->, line] ([yshift=2.5mm]plant.west)
    -- node[sig, above] {$P^{\mathrm{ref}}$} ([yshift=2.5mm]primary.east);
\draw[->, line] ([yshift=-2.5mm]plant.west)
    -- node[sig, below] {$Q^{\mathrm{ref}}$} ([yshift=-2.5mm]primary.east);
\node[signal, label={[sig]above:$P_{\mathrm{plant}}^{\mathrm{ref}}$}]
    (pref) at (25.6,-4.15) {};
\node[signal, label={[sig]below:$Q_{\mathrm{plant}}^{\mathrm{ref}}$}]
    (qref) at (25.6,-4.65) {};
\draw[->, line] (pref) -- ([yshift=2.5mm]plant.east);
\draw[->, line] (qref) -- ([yshift=-2.5mm]plant.east);
% Primary-control angle and frequency connections are implicit.
\else
\coordinate (controlextent) at (outer.south east);
\node[signal, label={[sig]above:$P^{\mathrm{ref}}$}] (pref) at (\setpointcolumn, -4.15) {};
\draw[->, line] (pref) -- ([yshift=2.5mm]outer.east);
\node[signal, label={[sig]below:$Q^{\mathrm{ref}}$}] (qref) at (\setpointcolumn, -4.65) {};
\draw[->, line] (qref) -- ([yshift=-2.5mm]outer.east);
\fi

% ================= Configuration and frame enclosures =================
\begin{scope}[on background layer]
    \node[operator enclosure, densely dotted,
        fit=(bridge)(zs)(zg)(cground)
            (icc)(outer)(pwm)(controlextent)] (enclosure) {};
    % Measurements and commands cross the reference-frame boundary.
    \coordinate (frame) at (\framecolumn,0 |- enclosure.center);
    \ifgfm
    \coordinate (frameedge) at ([xshift=3mm]outer.east);
    \draw[enclosure, densely dotted]
        (frame |- enclosure.south) -- (frame) -- (frame -| frameedge)
        -- (frameedge |- enclosure.south);
    \else
    \draw[enclosure, densely dotted]
        (frame |- enclosure.south) -- (frame) -- (frame -| enclosure.east);
    \fi
\end{scope}
\coordinate (framesymbol) at ($(frame)+(-0.65,0)$);
\node[sig] at ($(framesymbol)+(0,0.6)$) {$abc$};
\draw[->, line] ($(framesymbol)+(-0.12,0.32)$) -- ($(framesymbol)+(-0.12,-0.32)$);
\draw[->, line] ($(framesymbol)+(0.12,-0.32)$) -- ($(framesymbol)+(0.12,0.32)$);
\node[sig] at ($(framesymbol)+(0,-0.6)$) {$dq0$};

\end{tikzpicture}
