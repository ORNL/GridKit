% TUNABLES: gain width and spacing of the two lower branches.
\def\gainwidth{1.35cm}
\def\integralrow{-2.5}
\def\trackingrow{-4.2}

\begin{document}
\begin{tikzpicture}[emt diagram]

% Current limit and proportional path.
\node[block, minimum width=2.9cm, minimum height=1.3cm]
    (Ilim) at (-3,-0.8)
    {$\mathcal{L}_i^{-1/2}$};
\node[signal] (ilim) at (-1,-0.8) {};
\node[sum] (error) at (0,-0.8) {$\Sigma$};
\node[block, minimum width=\gainwidth] (Kp) at (2.2,-0.8) {$K_P$};
\node[sum] (command) at (6.3,-0.8) {$\Sigma$};
\draw[line] (Ilim.east) -- (ilim);
\draw[->, line] (ilim) -- (error.west);
\node[pm, above left=0pt and 1pt of error.west] {$+$};
\draw[->, line] (error.east) -- (Kp.west);
\draw[->, line] (Kp.east) -- (command.west);
\node[pm, above left=0pt and 1pt of command.west] {$+$};

% Integral path, with anti-windup added before integration.
\node[block, minimum width=\gainwidth] (Ki)
    at (0,\integralrow) {$K_I$};
\node[sum] (rate) at (2.2,\integralrow) {$\Sigma$};
\node[block, minimum width=\gainwidth] (integrator)
    at (6.3,\integralrow) {$\dfrac{1}{s}$};
\draw[->, line] (error.south) -- (Ki.north);
\draw[->, line] (Ki.east) -- (rate.west);
\node[pm, above left=0pt and 1pt of rate.west] {$+$};
\draw[->, line] (rate.east) -- (integrator.west);
\draw[->, line] (integrator.north) -- (command.south);
\node[pm, below right=1pt and 0pt of command.south] {$+$};

% Capacitor-voltage feedforward and synchronous-frame decoupling.
\node[block, minimum width=2.6cm]
    (feedforward) at (6.3,2.4)
    {$\mathbf{v}+\mathbf{J}\omega L\mathbf{i}$};
\draw[->, line] (feedforward.south)
    -- (command.north);
\node[pm, above right=1pt and 0pt of command.north] {$+$};

% Voltage command output and tracking return on the right boundary.
\node[signal] (u) at (7.8,-0.8) {};
\node[sum] (tracking) at (7.8,\trackingrow) {$\Sigma$};
\node[block, minimum width=\gainwidth] (Kaw)
    at (2.2,\trackingrow) {$K_{\mathrm{aw}}$};
\draw[line] (command.east) -- (u);
\draw[->, line] (u) -- node[sig, above, pos=0.75] {$\mathbf{u}$} (10.2,-0.8);
\draw[->, line] (u) -- (tracking.north);
\node[pm, above left=1pt and 0pt of tracking.north] {$-$};
\draw[->, line] (10.2,\trackingrow) -- node[sig, above, pos=0.25]
    {$\mathbf{u}^{\mathrm{lim}}$} (tracking.east);
\node[pm, above right=0pt and 1pt of tracking.east] {$+$};
\draw[->, line] (tracking.west) -- (Kaw.east);
\draw[->, line] (Kaw.north) -- (rate.south);
\node[pm, below left=1pt and 0pt of rate.south] {$+$};

% External command and measured quantities share the synchronous frame.
\node[sig] (icmd) at (-6,-0.8) {$\mathbf{i}^{\mathrm{cmd}}$};
\node[sig] (i) at (0,5) {$\mathbf{i}$};
\node[sig] (omega) at (6.3,5) {$\omega$};
\node[sig] (v) at (7.1,5) {$\mathbf{v}$};
\node[signal] (isplit) at (0,3.6) {};
\draw[->, line] (icmd.east) -- (Ilim.west);
\draw[->, line] (i.south) -- (error.north);
\node[pm, above right=1pt and 0pt of error.north] {$-$};
\draw[->, line] (isplit) -- (5.5,3.6) -- (feedforward.north -| 5.5,0);
\draw[->, line] (omega.south) -- (feedforward.north -| omega);
\draw[->, line] (v.south) -- (feedforward.north -| v);

% Limited current is returned to the outer controller.
\draw[->, line] (ilim) -- (-1,5);
\node[sig, above] at (-1,5) {$\mathbf{i}^{\mathrm{lim}}$};

% Enclose the model-owned states, variables, and feedback paths.
\begin{scope}[on background layer]
    \node[operator enclosure, densely dotted,
        fit=(Ilim)(ilim)(error)(Kp)(Ki)(rate)(integrator)
            (feedforward)(command)(u)(tracking)(Kaw)(isplit)] {};
\end{scope}

\end{tikzpicture}
